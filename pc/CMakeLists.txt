# Versión de PC (Windows) para probar sin compilar el APK. Usa las mismas fuentes de SDL que Android.
cmake_minimum_required(VERSION 3.16)
project(vascolet_pc C CXX)

set(JNI ${CMAKE_CURRENT_SOURCE_DIR}/../android/app/jni)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(${JNI}/SDL sdl)

set(BUILD_SHARED_LIBS OFF)
set(SDL2IMAGE_VENDORED OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_BACKEND_STB ON CACHE BOOL "" FORCE)
set(SDL2IMAGE_BACKEND_WIC OFF CACHE BOOL "" FORCE)
foreach(f AVIF BMP GIF JXL LBM PCX PNM QOI SVG TGA TIF WEBP XCF XPM XV)
    set(SDL2IMAGE_${f} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${JNI}/SDL_image sdl_image)

set(SDL2MIXER_VENDORED OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_DEPS_SHARED OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_VORBIS STB CACHE STRING "" FORCE)
foreach(f FLAC GME MOD MP3 MIDI OPUS WAVPACK CMD)
    set(SDL2MIXER_${f} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${JNI}/SDL_mixer sdl_mixer)

set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../src)
file(GLOB_RECURSE VC_SOURCES CONFIGURE_DEPENDS ${SRC}/*.cpp)
add_executable(VascoletCollection ${VC_SOURCES} ${JNI}/tinyxml2/tinyxml2.cpp)
target_compile_features(VascoletCollection PRIVATE cxx_std_17)
target_compile_definitions(VascoletCollection PRIVATE HOST _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_compile_options(VascoletCollection PRIVATE /utf-8 /W1)
target_include_directories(VascoletCollection PRIVATE ${SRC} ${JNI}/tinyxml2)
target_link_libraries(VascoletCollection PRIVATE SDL2::SDL2main SDL2_image::SDL2_image-static SDL2_mixer::SDL2_mixer-static SDL2::SDL2-static)
