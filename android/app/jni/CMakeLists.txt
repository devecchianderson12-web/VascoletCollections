cmake_minimum_required(VERSION 3.6)
project(xa C CXX)

set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(SDL)

set(BUILD_SHARED_LIBS OFF)
add_library(xa_sdl2_shim INTERFACE)
target_link_libraries(xa_sdl2_shim INTERFACE SDL2::SDL2)
add_library(SDL2::SDL2-static ALIAS xa_sdl2_shim)
set(SDL2IMAGE_VENDORED OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_BACKEND_STB ON CACHE BOOL "" FORCE)
foreach(f AVIF BMP GIF JXL LBM PCX PNM QOI SVG TGA TIF WEBP XCF XPM XV)
    set(SDL2IMAGE_${f} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(SDL_image)

set(SDL2MIXER_VENDORED OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_DEPS_SHARED OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_VORBIS STB CACHE STRING "" FORCE)
foreach(f FLAC GME MOD MP3 MIDI OPUS WAVPACK CMD)
    set(SDL2MIXER_${f} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(SDL_mixer)

set(XA_SRC ${CMAKE_CURRENT_SOURCE_DIR}/../../../src)
file(GLOB_RECURSE XA_SOURCES ${XA_SRC}/*.cpp)
add_library(main SHARED ${XA_SOURCES} tinyxml2/tinyxml2.cpp)
target_compile_features(main PRIVATE cxx_std_17)
target_compile_options(main PRIVATE -O2 -Wno-unused -Wno-format-security)
target_include_directories(main PRIVATE ${XA_SRC} tinyxml2)
target_link_libraries(main PRIVATE SDL2::SDL2main SDL2_image::SDL2_image-static SDL2_mixer::SDL2_mixer-static SDL2::SDL2 android log)
